if(ENABLE_TESTING)
  execute_process(
    COMMAND mlir-opt --emit-bytecode -o test_intr_MINIMUM.bc
            ${CMAKE_CURRENT_SOURCE_DIR}/test_intr_MINIMUM.mlir
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
  add_test(
    NAME INTR_MINIMUM
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMAND mli test_intr_MINIMUM.mlir --args=1.5,2.25)
  add_test(
    NAME INTR_MINIMUM_BYTECODE
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    COMMAND mli test_intr_MINIMUM.bc --args=1.5,2.25)
  set_tests_properties(
    INTR_MINIMUM INTR_MINIMUM_BYTECODE
    PROPERTIES TIMEOUT 60 LABELS "all" PASS_REGULAR_EXPRESSION "1.5")
  add_test(
    NAME INTR_MINIMUM_NAN
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMAND mli test_intr_MINIMUM.mlir --args=1,NaN)
  add_test(
    NAME INTR_MINIMUM_NAN_BYTECODE
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    COMMAND mli test_intr_MINIMUM.bc --args=1,NaN)
  set_tests_properties(
    INTR_MINIMUM_NAN INTR_MINIMUM_NAN_BYTECODE
    PROPERTIES TIMEOUT 60 LABELS "all" PASS_REGULAR_EXPRESSION "NaN")
endif()
